import requests from bs4 import BeautifulSoup import json import os import re from datetime import datetime # === KASA SINIFI === class Kasa: def __init__(self): self.dosya = "kasa_veri.json" self.urunler_dosya = "urunler.json" self.faturalar_dosya = "faturalar.json" self.bakiye = 0 self.urunler = [] self.faturalar = [] self.yukle() def yukle(self): # Kasa verileri if os.path.exists(self.dosya): with open(self.dosya, "r", encoding="utf-8") as f: veri = json.load(f) self.bakiye = veri.get("bakiye", 0) # Ürünler if os.path.exists(self.urunler_dosya): with open(self.urunler_dosya, "r", encoding="utf-8") as f: self.urunler = json.load(f) # Faturalar if os.path.exists(self.faturalar_dosya): with open(self.faturalar_dosya, "r", encoding="utf-8") as f: self.faturalar = json.load(f) def kaydet(self): with open(self.dosya, "w", encoding="utf-8") as f: json.dump({"bakiye": self.bakiye}, f, ensure_ascii=False, indent=2) with open(self.urunler_dosya, "w", encoding="utf-8") as f: json.dump(self.urunler, f, ensure_ascii=False, indent=2) with open(self.faturalar_dosya, "w", encoding="utf-8") as f: json.dump(self.faturalar, f, ensure_ascii=False, indent=2) # === ÜRÜN İŞLEMLERİ === def urun_ekle(self, ad, fiyat, stok=1): urun = { "id": len(self.urunler) + 1, "ad": ad, "fiyat": fiyat, "stok": stok, "eklenme_tarihi": datetime.now().strftime("%Y-%m-%d %H:%M") } self.urunler.append(urun) self.kaydet() return urun def urun_listele(self): if not self.urunler: print("📦 Hiç ürün yok.") return print("\n--- ÜRÜN LİSTESİ ---") for u in self.urunler: print(f"ID:{u['id']} | {u['ad']} | {u['fiyat']} TL | Stok:{u['stok']}") def urun_fiyat_guncelle(self, urun_id, yeni_fiyat): for u in self.urunler: if u['id'] == urun_id: u['fiyat'] = yeni_fiyat self.kaydet() print(f"✅ Ürün fiyatı güncellendi: {u['ad']} -> {yeni_fiyat} TL") return True print("❌ Ürün bulunamadı.") return False # === FATURA İŞLEMLERİ === def fatura_olustur(self, musteri_bilgileri, urun_ids): sepetteki_urunler = [] toplam = 0 for id in urun_ids: for u in self.urunler: if u['id'] == id: sepetteki_urunler.append(u.copy()) toplam += u['fiyat'] break if not sepetteki_urunler: print("❌ Sepet boş!") return fatura = { "fatura_no": len(self.faturalar) + 1, "tarih": datetime.now().strftime("%Y-%m-%d %H:%M"), "musteri": musteri_bilgileri, "urunler": sepetteki_urunler, "toplam": toplam, "durum": "Oluşturuldu" # Sonradan "Faturalandı" olarak değiştirilebilir } self.faturalar.append(fatura) self.bakiye += toplam self.kaydet() print(f"\n✅ FATURA OLUŞTURULDU!") print(f"Fatura No: {fatura['fatura_no']}") print(f"Toplam: {toplam} TL") print(f"Müşteri: {musteri_bilgileri['isim']}") print("Ürünler:") for u in sepetteki_urunler: print(f" - {u['ad']}: {u['fiyat']} TL") return fatura def faturalari_listele(self): if not self.faturalar: print("📄 Hiç fatura yok.") return print("\n--- FATURA LİSTESİ ---") for f in self.faturalar: print(f"No:{f['fatura_no']} | {f['tarih']} | {f['musteri']['isim']} | {f['toplam']} TL | {f['durum']}") def fatura_durum_guncelle(self, fatura_no, yeni_durum="Faturalandı"): for f in self.faturalar: if f['fatura_no'] == fatura_no: f['durum'] = yeni_durum self.kaydet() print(f"✅ Fatura {fatura_no} durumu güncellendi: {yeni_durum}") return True print("❌ Fatura bulunamadı.") return False # === MAKERPAZAR VERİ ÇEKİCİ === def makerpazar_urun_cek(url): try: headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"} response = requests.get(url, headers=headers, timeout=10) soup = BeautifulSoup(response.text, "html.parser") # Ürün adı ad_etiketi = soup.find(["h1", "h2", "span"], class_=re.compile("title|name|product-name")) ad = ad_etiketi.text.strip() if ad_etiketi else "Makerpazar Ürünü" # Fiyat fiyat_etiketi = soup.find(["span", "div", "p"], class_=re.compile("price|fiyat|product-price")) if fiyat_etiketi: fiyat_metin = fiyat_etiketi.text.strip() sayi = re.search(r"([\d,\.]+)", fiyat_metin.replace(".", "").replace(",", ".")) fiyat = float(sayi.group(1)) if sayi else 0 else: fiyat = 0 return {"ad": ad, "fiyat": fiyat} except Exception as e: print(f"❌ Veri çekme hatası: {e}") return None # === ANA PROGRAM === kasa = Kasa() while True: print("\n" + "="*50) print("🏪 MAKERPAZAR KASA & FATURA SİSTEMİ") print("="*50) print("1. Ürün Ekle (Manuel)") print("2. Makerpazar'dan Ürün Çek ve Ekle") print("3. Ürün Listele") print("4. Ürün Fiyatını Değiştir (Tek seferlik)") print("5. Fatura Oluştur") print("6. Faturaları Listele") print("7. Fatura Durumu Güncelle (Faturalandı/İptal)") print("8. Kasa Raporu") print("9. Çıkış") print("="*50) secim = input("Seçiminiz: ") if secim == "1": ad = input("Ürün adı: ") fiyat = float(input("Fiyat (TL): ")) stok = int(input("Stok adedi: ") or "1") kasa.urun_ekle(ad, fiyat, stok) print("✅ Ürün eklendi!") elif secim == "2": url = input("Makerpazar ürün URL'si (tam link): ") veri = makerpazar_urun_cek(url) if veri and veri['fiyat'] > 0: print(f"📦 {veri['ad']} - {veri['fiyat']} TL") onay = input("Bu ürünü eklemek istiyor musun? (e/h): ") if onay.lower() == "e": stok = int(input("Stok adedi: ") or "1") kasa.urun_ekle(veri['ad'], veri['fiyat'], stok) print("✅ Ürün eklendi!") else: print("❌ Ürün bulunamadı veya fiyat çekilemedi.") elif secim == "3": kasa.urun_listele() elif secim == "4": kasa.urun_listele() urun_id = int(input("Güncellemek istediğin ürün ID: ")) yeni_fiyat = float(input("Yeni fiyat (TL): ")) kasa.urun_fiyat_guncelle(urun_id, yeni_fiyat) elif secim == "5": # Müşteri bilgileri print("\n--- MÜŞTERİ BİLGİLERİ ---") musteri = { "isim": input("Müşteri adı: "), "vergi_no": input("Vergi No: "), "adres": input("Adres: "), "telefon": input("Telefon: "), "email": input("E-posta: ") } # Ürün seçimi kasa.urun_listele() urun_ids = [] while True: id = input("Sepete eklemek için ürün ID (bitirmek için '0'): ") if id == "0": break urun_ids.append(int(id)) if urun_ids: kasa.fatura_olustur(musteri, urun_ids) else: print("❌ Sepete ürün eklenmedi.") elif secim == "6": kasa.faturalari_listele() elif secim == "7": kasa.faturalari_listele() no = int(input("Güncellemek istediğin fatura no: ")) durum = input("Yeni durum (Faturalandı/İptal): ") kasa.fatura_durum_guncelle(no, durum) elif secim == "8": print(f"\n💰 KASA BAKİYESİ: {kasa.bakiye} TL") print(f"📦 Toplam Ürün: {len(kasa.urunler)}") print(f"📄 Toplam Fatura: {len(kasa.faturalar)}") elif secim == "9": print("Çıkılıyor... Veriler kaydedildi.") break else: print("Geçersiz seçim!")
